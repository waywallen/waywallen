cmake_minimum_required(VERSION 3.24)
project(wallpaper_filter_drafts_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Qml Protobuf ProtobufQuick)
qt_add_protobuf(filter_proto QML PROTO_FILES ../../../proto/filter.proto)
add_executable(filter_drafts_test filter_drafts_test.cpp)
target_include_directories(filter_drafts_test PRIVATE ../../include)
target_link_libraries(filter_drafts_test PRIVATE filter_proto Qt6::Core Qt6::Protobuf)
enable_testing()
add_test(NAME filter_drafts COMMAND filter_drafts_test)
